**Goal:** Build MeoInk — a FastAPI rendering server on pve01, a Flask display client on rpzero01, and a new "Ink" page in the meoctl dashboard for pushing images to the 800×480 Inky Impression e-ink display.

```python
from flask import Flask, request, jsonify
import os

app = Flask(__name__)
IMAGE_PATH = os.path.join(os.path.dirname(__file__), "current.png")

@app.post("/display")
def display():
    if "image" not in request.files:
        return jsonify({"error": "no image"}), 400

    image_bytes = request.files["image"].read()

    with open(IMAGE_PATH, "wb") as f:
        f.write(image_bytes)

    try:
        from PIL import Image
        from inky.auto import InkyAuto
        inky = InkyAuto()
        img = Image.open(IMAGE_PATH)
        inky.set_image(img, saturation=0.5)
        inky.show()
    except Exception as e:
        return jsonify({"error": str(e)}), 500

    return jsonify({"ok": True})

if __name__ == "__main__":
    app.run(host="0.0.0.0", port=5000)
```

```python
import io
import requests
from PIL import Image, ImageDraw, ImageFont
import pytz
from datetime import datetime

DISPLAY_SIZE = (800, 480)
BG_COLOR = (15, 15, 15)
TEXT_COLOR = (220, 220, 220)
DIM_COLOR = (120, 120, 120)
ACCENT_COLOR = (110, 231, 183)

WEATHER_CODES = {
    0: "Clear", 1: "Mostly Clear", 2: "Partly Cloudy", 3: "Overcast",
    45: "Foggy", 48: "Icy Fog",
    51: "Light Drizzle", 53: "Drizzle", 55: "Heavy Drizzle",
    61: "Light Rain", 63: "Rain", 65: "Heavy Rain",
    71: "Light Snow", 73: "Snow", 75: "Heavy Snow",
    80: "Rain Showers", 81: "Rain Showers", 82: "Heavy Showers",
    95: "Thunderstorm", 96: "Thunderstorm",
}

OPEN_METEO_URL = (
    "https://api.open-meteo.com/v1/forecast"
    "?latitude=39.758&longitude=-104.967"
    "&current=temperature_2m,apparent_temperature,weather_code,windspeed_10m,relative_humidity_2m"
    "&daily=weathercode,temperature_2m_max,temperature_2m_min"
    "&timezone=America%2FDenver"
    "&forecast_days=3"
    "&{units_params}"
)


def _font(size: int, bold: bool = False) -> ImageFont.FreeTypeFont:
    try:
        path = "/usr/share/fonts/truetype/dejavu/DejaVuSans{}.ttf".format(
            "-Bold" if bold else ""
        )
        return ImageFont.truetype(path, size)
    except Exception:
        return ImageFont.load_default()


def resize_image(image_bytes: bytes) -> bytes:
    img = Image.open(io.BytesIO(image_bytes)).convert("RGB")
    img.thumbnail(DISPLAY_SIZE, Image.LANCZOS)
    canvas = Image.new("RGB", DISPLAY_SIZE, BG_COLOR)
    x = (DISPLAY_SIZE[0] - img.width) // 2
    y = (DISPLAY_SIZE[1] - img.height) // 2
    canvas.paste(img, (x, y))
    buf = io.BytesIO()
    canvas.save(buf, "PNG")
    return buf.getvalue()
```

```python
class UrlRequest(BaseModel):
    url: str


@app.post("/render/upload")
async def render_upload(file: UploadFile = File(...)):
    data = await file.read()
    try:
        result = resize_image(data)
    except Exception as e:
        raise HTTPException(400, str(e))
    return Response(content=result, media_type="image/png")


@app.post("/render/url")
async def render_url(body: UrlRequest):
    import httpx
    async with httpx.AsyncClient(timeout=15) as client:
        try:
            r = await client.get(body.url)
            r.raise_for_status()
        except Exception as e:
            raise HTTPException(400, f"Failed to fetch URL: {e}")
    try:
        result = resize_image(r.content)
    except Exception as e:
        raise HTTPException(400, f"Not a valid image: {e}")
    return Response(content=result, media_type="image/png")
```

```python
def fetch_and_render_weather(units: str = "imperial") -> bytes:
    units_params = (
        "temperature_unit=fahrenheit&wind_speed_unit=mph"
        if units == "imperial"
        else "temperature_unit=celsius&wind_speed_unit=kmh"
    )
    url = OPEN_METEO_URL.format(units_params=units_params)
    r = requests.get(url, timeout=15)
    r.raise_for_status()
    return _render_weather_card(r.json(), units)


def _render_weather_card(data: dict, units: str) -> bytes:
    img = Image.new("RGB", DISPLAY_SIZE, BG_COLOR)
    d = ImageDraw.Draw(img)

    current = data["current"]
    daily = data["daily"]
    unit_sym = "°F" if units == "imperial" else "°C"
    speed_unit = "mph" if units == "imperial" else "km/h"

    temp = round(current["temperature_2m"])
    feels = round(current["apparent_temperature"])
    condition = WEATHER_CODES.get(current["weather_code"], "Unknown")
    wind = round(current["windspeed_10m"])
    humidity = current["relative_humidity_2m"]

    tz = pytz.timezone("America/Denver")
    now_str = datetime.now(tz).strftime("%-I:%M %p")

    # Header row
    d.text((30, 24), "Denver, CO", font=_font(18), fill=DIM_COLOR)
    d.text((770, 24), now_str, font=_font(18), fill=DIM_COLOR, anchor="ra")
    d.line([(30, 52), (770, 52)], fill=(40, 40, 40), width=1)

    # Current temp (large)
    d.text((30, 65), f"{temp}{unit_sym}", font=_font(96, bold=True), fill=TEXT_COLOR)

    # Current conditions (right of temp)
    d.text((380, 90), condition, font=_font(32, bold=True), fill=ACCENT_COLOR)
    d.text((380, 136), f"Feels like {feels}{unit_sym}", font=_font(22), fill=DIM_COLOR)
    d.text((380, 168), f"Wind {wind} {speed_unit}  ·  Humidity {humidity}%",
           font=_font(20), fill=DIM_COLOR)

    # Divider before forecast
    d.line([(30, 240), (770, 240)], fill=(40, 40, 40), width=1)

    # 3-day forecast
    days = ["Today", "Tomorrow", _day_name(daily["time"][2])]
    col_w = DISPLAY_SIZE[0] // 3
    for i in range(3):
        cx = i * col_w + col_w // 2
        hi = round(daily["temperature_2m_max"][i])
        lo = round(daily["temperature_2m_min"][i])
        cond = WEATHER_CODES.get(daily["weathercode"][i], "Unknown")
        d.text((cx, 268), days[i], font=_font(20, bold=True), fill=TEXT_COLOR, anchor="mm")
        d.text((cx, 312), f"H: {hi}{unit_sym}  L: {lo}{unit_sym}",
               font=_font(18), fill=DIM_COLOR, anchor="mm")
        d.text((cx, 348), cond, font=_font(16), fill=DIM_COLOR, anchor="mm")
        if i < 2:
            d.line([(col_w * (i + 1), 255), (col_w * (i + 1), 375)],
                   fill=(40, 40, 40), width=1)

    buf = io.BytesIO()
    img.save(buf, "PNG")
    return buf.getvalue()


def _day_name(date_str: str) -> str:
    return datetime.strptime(date_str, "%Y-%m-%d").strftime("%A")
```

```tsx
// dashboard/src/client/components/InkPage.tsx
import React, { useState, useRef, useEffect, useCallback } from 'react';

const MEOINK = `http://${window.location.hostname}:8183`;
const DISPLAY_W = 800;
const DISPLAY_H = 480;

type Tab = 'upload' | 'weather' | 'url';

interface CurrentInfo {
  timestamp: string;
  type: string;
  image_url: string;
}

export default function InkPage() {
  const [tab, setTab] = useState<Tab>('upload');
  const [previewUrl, setPreviewUrl] = useState<string | null>(null);
  const [pushing, setPushing] = useState(false);
  const [pushStatus, setPushStatus] = useState<'idle' | 'ok' | 'error'>('idle');
  const [current, setCurrent] = useState<CurrentInfo | null>(null);
  const [loading, setLoading] = useState(false);
  const [error, setError] = useState<string | null>(null);

  const fetchCurrent = useCallback(() => {
    fetch(`${MEOINK}/current`)
      .then(r => r.ok ? r.json() : null)
      .then(data => { if (data) setCurrent(data); })
      .catch(() => {});
  }, []);

  useEffect(() => {
    fetchCurrent();
  }, [pushStatus, fetchCurrent]);

  function switchTab(t: Tab) {
    setTab(t);
    setPreviewUrl(null);
    setError(null);
    setPushStatus('idle');
  }

  async function handlePush() {
    if (!previewUrl) return;
    setPushing(true);
    setPushStatus('idle');
    try {
      const blob = await (await fetch(previewUrl)).blob();
      const form = new FormData();
      form.append('file', blob, 'display.png');
      form.append('content_type', tab);
      const r = await fetch(`${MEOINK}/push`, { method: 'POST', body: form });
      setPushStatus(r.ok ? 'ok' : 'error');
    } catch {
      setPushStatus('error');
    } finally {
      setPushing(false);
    }
  }

  return (
    <div style={{ maxWidth: 480, margin: '0 auto', padding: '0 0 80px 0' }}>
      <div style={{ padding: '16px 16px 0' }}>
        <h2 style={{ margin: 0, fontSize: 18, fontWeight: 700 }}>Ink</h2>
        <p style={{ margin: '4px 0 0', fontSize: 13, color: 'var(--text-dim, #888)' }}>
          Push content to the e-ink display
        </p>
      </div>

      {/* Tabs */}
      <div style={{ display: 'flex', borderBottom: '1px solid var(--border, #2a2a2a)', marginTop: 16 }}>
        {(['upload', 'weather', 'url'] as Tab[]).map(t => (
          <button key={t} onClick={() => switchTab(t)} style={{
            flex: 1, padding: '10px 4px', border: 'none', background: 'transparent',
            color: tab === t ? 'var(--accent, #6ee7b7)' : 'var(--text-dim, #888)',
            borderBottom: tab === t ? '2px solid var(--accent, #6ee7b7)' : '2px solid transparent',
            cursor: 'pointer', fontSize: 13, fontWeight: tab === t ? 600 : 400,
          }}>
            {t === 'upload' ? '↑ Upload' : t === 'weather' ? '☁ Weather' : '🔗 URL'}
          </button>
        ))}
      </div>

      {/* Tab panels */}
      <div style={{ padding: 16 }}>
        {tab === 'upload' && (
          <UploadPanel
            onPreview={setPreviewUrl}
            onError={setError}
            setLoading={setLoading}
          />
        )}
        {tab === 'weather' && (
          <WeatherPanel
            onPreview={setPreviewUrl}
            onError={setError}
            setLoading={setLoading}
          />
        )}
        {tab === 'url' && (
          <UrlPanel
            onPreview={setPreviewUrl}
            onError={setError}
            setLoading={setLoading}
          />
        )}

        {/* Preview */}
        {previewUrl && (
          <div style={{ marginTop: 16 }}>
            <div style={{ fontSize: 11, color: 'var(--text-dim, #888)', marginBottom: 6, textTransform: 'uppercase', letterSpacing: 1 }}>
              Preview — 800×480
            </div>
            <img
              src={previewUrl}
              alt="Preview"
              style={{ width: '100%', aspectRatio: `${DISPLAY_W}/${DISPLAY_H}`, borderRadius: 6, border: '1px solid var(--border, #2a2a2a)' }}
            />
          </div>
        )}

        {error && (
          <p style={{ color: '#f87171', fontSize: 13, marginTop: 12 }}>{error}</p>
        )}

        {/* Push button */}
        <button
          onClick={handlePush}
          disabled={!previewUrl || pushing}
          style={{
            marginTop: 16, width: '100%', padding: '11px', border: 'none', borderRadius: 6,
            background: previewUrl ? 'var(--accent, #6ee7b7)' : 'var(--surface2, #2a2a2a)',
            color: previewUrl ? '#000' : 'var(--text-dim, #888)',
            fontWeight: 700, fontSize: 14, cursor: previewUrl ? 'pointer' : 'not-allowed',
          }}
        >
          {pushing ? 'Pushing…' : pushStatus === 'ok' ? '✓ Pushed' : pushStatus === 'error' ? '✗ Push failed' : 'Push to Display'}
        </button>
      </div>

      {/* Current display strip */}
      {current && (
        <div style={{
          position: 'fixed', bottom: 0, left: 0, right: 0,
          background: 'var(--surface, #111)', borderTop: '1px solid var(--border, #2a2a2a)',
          padding: '10px 16px', display: 'flex', alignItems: 'center', gap: 12,
        }}>
          <img
            src={`${MEOINK}/current/image.png`}
            alt="Current display"
            style={{ width: 48, height: 29, borderRadius: 3, objectFit: 'cover', border: '1px solid var(--border, #2a2a2a)' }}
          />
          <div>
            <div style={{ fontSize: 12, color: 'var(--text, #ccc)' }}>Current display</div>
            <div style={{ fontSize: 11, color: 'var(--text-dim, #888)' }}>
              rpzero01 · {current.type} · {new Date(current.timestamp).toLocaleTimeString()}
            </div>
          </div>
          <div style={{ marginLeft: 'auto', width: 8, height: 8, background: '#6ee7b7', borderRadius: '50%' }} />
        </div>
      )}
    </div>
  );
}

// --- Sub-panels ---

function UploadPanel({ onPreview, onError, setLoading }: {
  onPreview: (url: string) => void;
  onError: (msg: string) => void;
  setLoading: (v: boolean) => void;
}) {
  const inputRef = useRef<HTMLInputElement>(null);

  async function handleFile(file: File) {
    setLoading(true);
    onError('');
    try {
      const form = new FormData();
      form.append('file', file);
      const r = await fetch(`${MEOINK}/render/upload`, { method: 'POST', body: form });
      if (!r.ok) throw new Error(await r.text());
      const blob = await r.blob();
      onPreview(URL.createObjectURL(blob));
    } catch (e: any) {
      onError(e.message ?? 'Upload failed');
    } finally {
      setLoading(false);
    }
  }

  return (
    <div
      onClick={() => inputRef.current?.click()}
      onDragOver={e => e.preventDefault()}
      onDrop={e => { e.preventDefault(); const f = e.dataTransfer.files[0]; if (f) handleFile(f); }}
      style={{
        border: '1.5px dashed var(--border, #333)', borderRadius: 8, padding: '32px 16px',
        textAlign: 'center', cursor: 'pointer', color: 'var(--text-dim, #888)',
      }}
    >
      <div style={{ fontSize: 32, marginBottom: 8 }}>📁</div>
      <div style={{ fontSize: 13 }}>Drop an image or tap to choose</div>
      <input ref={inputRef} type="file" accept="image/*" style={{ display: 'none' }}
        onChange={e => { const f = e.target.files?.[0]; if (f) handleFile(f); }} />
    </div>
  );
}

function WeatherPanel({ onPreview, onError, setLoading }: {
  onPreview: (url: string) => void;
  onError: (msg: string) => void;
  setLoading: (v: boolean) => void;
}) {
  const [units, setUnits] = useState<'imperial' | 'metric'>('imperial');

  async function handleRender() {
    setLoading(true);
    onError('');
    try {
      const r = await fetch(`${MEOINK}/render/weather`, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ units }),
      });
      if (!r.ok) throw new Error(await r.text());
      const blob = await r.blob();
      onPreview(URL.createObjectURL(blob));
    } catch (e: any) {
      onError(e.message ?? 'Render failed');
    } finally {
      setLoading(false);
    }
  }

  return (
    <div style={{ background: 'var(--surface2, #1a1a1a)', borderRadius: 8, padding: 14, border: '1px solid var(--border, #2a2a2a)' }}>
      <div style={{ display: 'flex', gap: 12, alignItems: 'center', marginBottom: 12 }}>
        <div>
          <div style={{ fontSize: 11, color: 'var(--text-dim, #888)', marginBottom: 2 }}>Location</div>
          <div style={{ fontSize: 13, color: 'var(--text, #ccc)' }}>80205 — Denver, CO</div>
        </div>
        <div style={{ marginLeft: 'auto' }}>
          <div style={{ fontSize: 11, color: 'var(--text-dim, #888)', marginBottom: 2 }}>Units</div>
          <select
            value={units}
            onChange={e => setUnits(e.target.value as 'imperial' | 'metric')}
            style={{ background: 'var(--surface, #111)', border: '1px solid var(--border, #333)', borderRadius: 4, padding: '4px 8px', color: 'var(--text, #ccc)', fontSize: 13 }}
          >
            <option value="imperial">Imperial</option>
            <option value="metric">Metric</option>
          </select>
        </div>
      </div>
      <button onClick={handleRender} style={{
        width: '100%', padding: 8, background: 'var(--surface, #111)',
        border: '1px solid var(--border, #444)', borderRadius: 6,
        color: 'var(--text, #ccc)', fontSize: 13, cursor: 'pointer',
      }}>
        Render Preview
      </button>
    </div>
  );
}

function UrlPanel({ onPreview, onError, setLoading }: {
  onPreview: (url: string) => void;
  onError: (msg: string) => void;
  setLoading: (v: boolean) => void;
}) {
  const [url, setUrl] = useState('');

  async function handleFetch() {
    if (!url) return;
    setLoading(true);
    onError('');
    try {
      const r = await fetch(`${MEOINK}/render/url`, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ url }),
      });
      if (!r.ok) throw new Error(await r.text());
      const blob = await r.blob();
      onPreview(URL.createObjectURL(blob));
    } catch (e: any) {
      onError(e.message ?? 'Fetch failed');
    } finally {
      setLoading(false);
    }
  }

  return (
    <div style={{ background: 'var(--surface2, #1a1a1a)', borderRadius: 8, padding: 14, border: '1px solid var(--border, #2a2a2a)' }}>
      <div style={{ fontSize: 11, color: 'var(--text-dim, #888)', marginBottom: 6 }}>Image URL</div>
      <div style={{ display: 'flex', gap: 8 }}>
        <input
          value={url}
          onChange={e => setUrl(e.target.value)}
          onKeyDown={e => e.key === 'Enter' && handleFetch()}
          placeholder="https://..."
          style={{
            flex: 1, background: 'var(--surface, #111)', border: '1px solid var(--border, #333)',
            borderRadius: 6, padding: '7px 10px', color: 'var(--text, #ccc)', fontSize: 13,
          }}
        />
        <button onClick={handleFetch} style={{
          padding: '7px 14px', background: 'var(--surface, #111)',
          border: '1px solid var(--border, #444)', borderRadius: 6,
          color: 'var(--text, #ccc)', fontSize: 13, cursor: 'pointer',
        }}>
          Fetch
        </button>
      </div>
    </div>
  );
}
```
